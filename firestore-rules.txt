rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn() { return request.auth != null && request.auth.token.email_verified == true; }
    function isAdmin()  { return signedIn() && request.auth.token.email == 'as6064147@gmail.com'; }
    function isEditor() { return signedIn() && exists(/databases/$(database)/documents/roles/$(request.auth.token.email)); }

    // إعدادات شكل الجريدة: الكل يقرا، المدير بس يعدّل
    match /site/{doc} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // قائمة المحررين: المدير بس يضيف ويشيل
    match /roles/{email} {
      allow read: if isAdmin() || (signedIn() && request.auth.token.email == email);
      allow write: if isAdmin();
    }

    // الأخبار: الكل يقرا، المحرر يتحكم في أخباره بس، المدير في كل حاجة
    match /articles/{id} {
      allow read: if true;
      allow create: if isAdmin() || (isEditor()
        && request.resource.data.authorEmail == request.auth.token.email
        && request.resource.data.get('pinned', false) == false);
      allow update: if isAdmin() || (isEditor()
        && resource.data.authorEmail == request.auth.token.email
        && request.resource.data.authorEmail == resource.data.authorEmail
        && request.resource.data.get('pinned', false) == resource.data.get('pinned', false));
      allow delete: if isAdmin() || (isEditor() && resource.data.authorEmail == request.auth.token.email);
    }
  }
}
