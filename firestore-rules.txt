rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    match /site/{doc} {
      allow read: if true;
      allow write: if true;
    }

    match /roles/{email} {
      allow read: if true;
      allow write: if true;
    }

    match /articles/{id} {
      allow read: if true;
      allow create: if true;
      allow update: if true;
      allow delete: if true;
    }
  }
}
